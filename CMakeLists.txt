cmake_minimum_required(VERSION 3.20)
project(NetSyncLab VERSION 0.1.0 LANGUAGES CXX)

add_library(netsync_core src/model.cpp src/simulation.cpp)
target_include_directories(netsync_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(netsync_core PUBLIC cxx_std_20)

add_executable(netsync src/main.cpp)
target_link_libraries(netsync PRIVATE netsync_core)

if(MSVC)
  target_compile_options(netsync_core PRIVATE /W4 /WX)
  target_compile_options(netsync PRIVATE /W4 /WX)
else()
  target_compile_options(netsync_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(netsync PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(netsync_tests tests/test_main.cpp)
  target_link_libraries(netsync_tests PRIVATE netsync_core)
  if(MSVC)
    target_compile_options(netsync_tests PRIVATE /W4 /WX)
  else()
    target_compile_options(netsync_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME unit_tests COMMAND netsync_tests)
  add_test(NAME demo_converges COMMAND netsync demo)
  add_test(NAME ideal_converges COMMAND netsync ideal)
  add_test(NAME trace_converges COMMAND netsync trace)
  add_test(NAME cli_golden_output COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:netsync>")
endif()
